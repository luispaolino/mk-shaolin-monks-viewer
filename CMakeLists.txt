cmake_minimum_required(VERSION 3.25)
project(ShaolinMonksViewer LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The standalone viewer requires 64-bit Windows")
endif()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(INSTALL_PROJECT OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/third_party/miniz" "${CMAKE_BINARY_DIR}/miniz")
add_library(viewer_assets STATIC "${ROOT}/src/viewer/assets.cpp" "${ROOT}/src/viewer/animation.cpp" "${ROOT}/src/viewer/scenery.cpp" "${ROOT}/src/viewer/particles.cpp" "${ROOT}/src/viewer/geometry.cpp" "${ROOT}/src/viewer/art.cpp" "${ROOT}/src/viewer/atmosphere.cpp" "${ROOT}/third_party/xbox-disc-reader/kernel/disc.cpp")
target_include_directories(viewer_assets PUBLIC "${ROOT}/src/viewer" "${ROOT}/third_party/xbox-disc-reader")
target_sources(viewer_assets PRIVATE "${ROOT}/src/viewer/character_debug.cpp")
target_sources(viewer_assets PRIVATE "${ROOT}/src/viewer/action_graph.cpp")
target_sources(viewer_assets PRIVATE "${ROOT}/src/viewer/music.cpp" "${ROOT}/src/viewer/music_player.cpp")
target_sources(viewer_assets PRIVATE "${ROOT}/src/viewer/fbx.cpp")
target_compile_definitions(viewer_assets PRIVATE MINIZ_NO_ZLIB_COMPATIBLE_NAMES NOMINMAX UNICODE _UNICODE)
target_link_libraries(viewer_assets PUBLIC miniz xaudio2)
set(IMGUI "${ROOT}/third_party/imgui")
add_library(viewer_imgui OBJECT "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp" "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp" "${IMGUI}/backends/imgui_impl_win32.cpp" "${IMGUI}/backends/imgui_impl_dx11.cpp")
target_include_directories(viewer_imgui PUBLIC "${IMGUI}")
target_compile_definitions(viewer_imgui PUBLIC NOMINMAX UNICODE _UNICODE IMGUI_IMPL_WIN32_DISABLE_GAMEPAD)
add_executable(ShaolinMonksViewer "${ROOT}/src/viewer/main.cpp")
target_include_directories(ShaolinMonksViewer PRIVATE "${IMGUI}")
target_compile_definitions(ShaolinMonksViewer PRIVATE NOMINMAX UNICODE _UNICODE IMGUI_IMPL_WIN32_DISABLE_GAMEPAD)
target_link_libraries(ShaolinMonksViewer PRIVATE viewer_assets viewer_imgui d3d11 dxgi d3dcompiler dwmapi user32 gdi32 shell32 comdlg32)
target_link_options(ShaolinMonksViewer PRIVATE "LINKER:/SUBSYSTEM:WINDOWS" "LINKER:/ENTRY:wmainCRTStartup")
enable_testing()
